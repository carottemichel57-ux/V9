cmake_minimum_required(VERSION 3.5)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(DEFINED ENV{VITASDK})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "VitaSDK toolchain")
  else()
    message(FATAL_ERROR "VITASDK is not defined")
  endif()
endif()
project(saity_kernel C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wl,-q -O2 -std=gnu99")
add_executable(saity_kernel saity_kernel.c)
target_link_libraries(saity_kernel taihen_stub SceSysmemForDriver_stub SceIofilemgrForDriver_stub SceSysclibForDriver_stub)
set_target_properties(saity_kernel PROPERTIES LINK_FLAGS "-nostdlib")
vita_create_self(saity_kernel.skprx saity_kernel CONFIG ${CMAKE_CURRENT_SOURCE_DIR}/exports.yml UNSAFE)
vita_create_stubs(saity_kernel-stubs saity_kernel ${CMAKE_CURRENT_SOURCE_DIR}/exports.yml KERNEL)
